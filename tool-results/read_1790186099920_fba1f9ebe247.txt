     3→// Used by /api/setup-db (manual) and /api/health (auto-heal when tables are
     4→// missing) so a freshly-swapped database repairs itself instead of 500ing
     5→// every API (the "الفديو مش شغال" outage class).
     6→import { createClient } from '@libsql/client'
     7→
     8→export function makeLibsqlClient() {
     9→  var dbUrl = process.env.TURSO_DATABASE_URL || process.env.DATABASE_URL || ''
    10→  var authToken = process.env.TURSO_AUTH_TOKEN || process.env.DATABASE_AUTH_TOKEN || ''
    11→  if (!dbUrl) return null
    12→  return createClient({ url: dbUrl, authToken: authToken || undefined })
    13→}

   281→/* ============================================================
   282→ * 2026-و23 — **إصلاح بطء المنصة** (طلب المستر: «المنصة بطيئة، تسجيل
   283→ * الدخول وطلابي بياخدوا وقت عقبال ما يحملوا»):
   284→ * الجذر: كل إقلاع سيرفر (cold start على Vercel) كان بيشغّل الدورة
   285→ * الكاملة: 48 محاولة ALTER TABLE (بتفشل كلها بـ duplicate) + ~21
   286→ * UPDATE — يعني ~70 استعلام متسلسل على Turso بيضيفوا 2-4 ثواني على
   287→ * أول طلب بعد كل إقلاع. الحل: بصمة (هاش) لبنية الجداول والأعمدة
   288→ * والفهارس متخزنة في SiteConfig — لو البصمة مطابقة → تخطي كل حاجة
   289→ * (استعلام واحد بس)، ولو اتغيرت البنية مستقبلًا → البصمة تتغير
   290→ * والترميم بيشغّل لوحده من غير صيانة يدوية.
   291→ * ============================================================ */
   292→import { createHash } from 'crypto'
   293→
   294→/* ============================================================
   295→ * 2026-و30 — تسريع أول دخول (طلب المستر: «لما يجي يدخل الطفل الأولاني
   296→ * بيقعد وقت... لو تقدر سرعه»):
   297→ * (1) الترميم الكامل كان ~80 استعلام متسلسل على Turso (كل واحد =
   298→ *     roundtrip شبكة) = 10-14 ثانية على أول طلب بعد كل نشر — بقوا
   299→ *     **متوازيين** على دفعات = ~2 ثانية.
   300→ * (2) ميمو على مستوى الموديول: بعد أول نجاح في نفس الـ instance
   301→ *     مفيش حتى استعلام البصمة — الدوال بترجع فورًا.
   302→ * ============================================================ */
   303→var _schemaVerifiedInProcess = false
   304→
   305→function currentSchemaHash(): string {
   306→  var joined = SCHEMA_TABLES.join('||') + '##' +
   307→    SCHEMA_COLUMNS.map(function (c) { return c.join('.') }).join('|') + '##' +
   308→    SCHEMA_FIXES.join('##') + '##' +
   309→    SCHEMA_INDEXES.join('##')
   310→  return createHash('md5').update(joined).digest('hex').substring(0, 12)
   311→}

   313→/* تنفيذ استعلام متسامح: بيجرب مرتين (الدفعة المتوازية ممكن تصطدم بـ
   314→   busy لحظي) وبيتجاهل duplicate/already exists — والفشل الحقيقي
   315→   بيتسجل في النتايج من غير ما يبوّظ الباقي */
   316→async function execTolerant(client: any, sql: string, meta: any, results: any[]) {
   317→  for (var attempt = 0; attempt < 2; attempt++) {
   318→    try {
   319→      await client.execute(sql)
   320→      if (results) results.push(Object.assign({ ok: true }, meta))
   321→      return
   322→    } catch (e: any) {
   323→      var msg = String((e && e.message) || '')
   324→      if (msg.indexOf('duplicate') !== -1 || msg.indexOf('already exists') !== -1) return
   325→      if (attempt === 0) { await new Promise(function (r) { setTimeout(r, 250) }); continue }
   326→      if (results) results.push(Object.assign({ ok: false, error: msg }, meta))
   327→    }
   328→  }
   329→}

   331→export async function ensureSchema(client: any, opts?: { force?: boolean }) {
   332→  var force = !!(opts && opts.force)
   333→  var results: any[] = []
   334→
   335→  /* المسار الأسرع: الـ instance ده اتأكد من السكيما قبل كده → صفر استعلامات */
   336→  if (!force && _schemaVerifiedInProcess) {
   337→    return { missing: [], repaired: false, skipped: true, memo: true, results: [] }
   338→  }
   339→
   340→  /* المسار السريع: البصمة متخزنة ومطابقة → مفيش أي ترميم محتاج
   341→     (استعلام واحد بدل ~70 — ده اللي هيخلي الدخول ولوحة الطلاب فورًا) */
   342→  if (!force) {
   343→    try {
   344→      var flagRes = await client.execute({
   345→        sql: 'SELECT value FROM SiteConfig WHERE key = ? LIMIT 1',
   346→        args: [SCHEMA_HASH_KEY],
   347→      })
   348→      var stored = flagRes && flagRes.rows && flagRes.rows.length > 0 ? String(flagRes.rows[0].value || '') : ''
   349→      if (stored && stored === currentSchemaHash()) {
   350→        _schemaVerifiedInProcess = true
   351→        return { missing: [], repaired: false, skipped: true, results: [] }
   352→      }
   353→    } catch (e) {
   354→      // لو الجدول نفسه مش موجود (قاعدة جديدة) → الدورة الكاملة تحت
   355→    }
   356→  }
   357→
   358→  // Which core tables already exist?
   359→  var existing: string[] = []
   360→  try {
   361→    var res = await client.execute("SELECT name FROM sqlite_master WHERE type='table'")
   362→    for (var i = 0; i < res.rows.length; i++) existing.push(String(res.rows[i].name))
   363→  } catch (e) {}
   364→
   365→  var missing = CORE_TABLES.filter(function (t) { return existing.indexOf(t) === -1 })
   366→
   367→  // Only run DDL when something is actually missing (or when forced by explicit setup)
   368→  var tablesToRun = (missing.length > 0 || force) ? SCHEMA_TABLES : []
   369→  // (2026-و30) الجداول متوازية — مستقلة عن بعض (IF NOT EXISTS)
   370→  await Promise.all(tablesToRun.map(function (sql) {
   371→    var name = sql.match(/CREATE TABLE IF NOT EXISTS (\w+)/)?.[1]
   372→    return execTolerant(client, sql, { table: name }, results)
   373→  }))
   374→
   375→  // Columns (tolerant of duplicates) — (2026-و30) على دفعات متوازية بدل تسلسلي
   376→  var CHUNK = 8
   377→  for (var k = 0; k < SCHEMA_COLUMNS.length; k += CHUNK) {
   378→    await Promise.all(SCHEMA_COLUMNS.slice(k, k + CHUNK).map(function (c) {
   379→      var sql = 'ALTER TABLE ' + c[0] + ' ADD COLUMN ' + c[1] + ' ' + c[2] + ' ' + c[3]
   380→      return execTolerant(client, sql, { table: c[0], column: c[1] }, results)
   381→    }))
   382→  }
   383→
   384→  // NULL fixes — (2026-و30) متوازية (كلها idempotent)
   385→  await Promise.all(SCHEMA_FIXES.map(function (sql) {
   386→    return client.execute(sql).catch(function () {})
   387→  }))
   388→
   389→  // Indexes (idempotent — CREATE INDEX IF NOT EXISTS) — متوازية
   390→  await Promise.all(SCHEMA_INDEXES.map(function (sql) {
   391→    return execTolerant(client, sql, { index: sql }, results)
   392→  }))
   393→
   394→  _schemaVerifiedInProcess = true
   395→
   396→  // تخزين بصمة البنية — الإقلاعات الجاية بتتخطى الترميم كله
   397→  try {
   398→    var hash = currentSchemaHash()
   399→    try {
   400→      await client.execute({ sql: 'UPDATE SiteConfig SET value = ?, updatedAt = CURRENT_TIMESTAMP WHERE key = ?', args: [hash, SCHEMA_HASH_KEY] })
   401→    } catch (uErr) {
   402→      try {
   403→        await client.execute({ sql: 'INSERT INTO SiteConfig (id, key, value) VALUES (?, ?, ?)', args: ['sch' + hash + Date.now().toString(36), SCHEMA_HASH_KEY, hash] })
   404→      } catch (iErr) {}
   405→    }
   406→  } catch (hErr) {}
   407→
   408→  return { missing, repaired: missing.length > 0, results }
   409→}
