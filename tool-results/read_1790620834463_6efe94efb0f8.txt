    24→    24→function toBase64(file: File): Promise<string> {

    30→    30→function getMimeType(file: File): string {
    31→    31→  var fname = (file.name || '').toLowerCase()
    32→    32→  if (fname.endsWith('.pdf')) return 'application/pdf'
    33→    33→  if (fname.endsWith('.png')) return 'image/png'
    34→    34→  if (fname.endsWith('.webp')) return 'image/webp'
    35→    35→  return file.type || 'image/jpeg'
    36→    36→}

    67→    67→/*
    68→    68→ * normalizeMath — server-side cleanup of AI math output so the stored text
    69→    69→ * matches the canonical platform format that FractionText renders:
    70→    70→ *   - strip $…$ / $$…$$ wrappers
    71→    71→ *   - \\left( \\right) → plain parens
    72→    72→ *   - \\frac{(X)}{(Y)} → \\frac{X}{Y} (only when ONE paren pair wraps the
    73→    73→ *     whole numerator/denominator — (a+1)(a-1) stays untouched)
    74→    74→ */
    75→    75→function normalizeMath(s: string): string {
    76→    76→  if (!s) return s
    77→    77→  // first undo any JSON-escape corruption (\f eaten → "rac" …)
    78→    78→  var out = repairCorruptMath(String(s))
    79→    79→  out = out.replace(/\$\$([\s\S]+?)\$\$/g, '$1').replace(/\$([^$\n]+?)\$/g, '$1')
    80→    80→  out = out.replace(/\\left\s*/g, '').replace(/\\right\s*/g, '')
    81→    81→  out = out.replace(/\\frac\s*\{\s*\(([^{}]*)\)\s*\}\s*\{\s*\(([^{}]*)\)\s*\}/g, '\\frac{$1}{$2}')
    82→    82→  return out
    83→    83→}

   376→   376→function finalizeExtracted(extracted: any, type: string, grade: string, twoFilesMode: boolean, figuresCrop?: any): any {
   377→   377→  if (!extracted.title) { extracted.title = type + ' - ' + grade }

   379→   379→  if (!Array.isArray(extracted.questions)) { extracted.questions = [] }
   380→   380→  if (!extracted.answerKey) { extracted.answerKey = '' }
   381→   381→
   382→   382→  extracted.questions = extracted.questions.map(function(q) {
   383→   383→    /* (و45) سؤال له اختيارات (نص أو صور/رسومات) = اختياري دايمًا — ممنوع يتحول مقالي */
   384→   384→    var qType = isWritingQuestion(q) ? 'writing' : 'mcq'
   385→   385→    /* (2026-و40-w) حقول ورقة العمل — pass-through (المصدر: البحث عن الجداول
   386→   386→       والرسومات في ورقة المستر): sourcePage/srcName/table/figure/optionFigures
   387→   387→       بتتحفظ زي ما هي جنب الحقول القانونية — مع تطبيع خفيف */
   388→   388→    var worksheet: any = {}
   389→   389→    var spN = parseInt(String(q.sourcePage), 10)
   390→   390→    if (isFinite(spN) && spN > 0) worksheet.sourcePage = spN
   391→   391→    if (q.srcName && String(q.srcName).trim()) worksheet.srcName = String(q.srcName).trim()
   392→   392→    if (q.table && Array.isArray(q.table.rows)) worksheet.table = q.table
   393→   393→    if (q.figure && q.figure.bbox) worksheet.figure = q.figure
   394→   394→    if (Array.isArray(q.optionFigures)) worksheet.optionFigures = q.optionFigures
   395→   395→    var withWs = function (obj: any) { return Object.assign({}, obj, worksheet) }
   396→   396→    if (qType === 'writing') {
   397→   397→      return withWs({
   398→   398→        type: 'writing',
   399→   399→        question: normalizeMath(q.question || ''),
   400→   400→        options: [],
   401→   401→        correct: -1,
   402→   402→        points: q.points || 5,
   403→   403→        modelAnswer: normalizeMath(q.modelAnswer || q.answer || ''),
   404→   404→        acceptedAnswers: (Array.isArray(q.acceptedAnswers) ? q.acceptedAnswers : []).map(function(a: any) { return normalizeMath(String(a)) })

   408→   408→      type: 'mcq',
