   199→function battleStore(): Storage | null {
   200→  try { return window.sessionStorage || null } catch (e) { return null }

   203→function saveMeLocal(code: string, me: { id: string; token: string }): void {
   204→  try {
   205→    var st = battleStore()
   206→    if (st) st.setItem(ME_KEY_PREFIX + code, JSON.stringify({ playerId: me.id, token: me.token, ts: Date.now() }))
   207→  } catch (e) { /* التخزين ممكن يكون مقفول — مش مشكلة قاطعة */ }
   208→}
   209→
   210→function clearMeLocal(code: string): void {
   211→  try { var st = battleStore(); if (st) st.removeItem(ME_KEY_PREFIX + code) } catch (e) {}
   212→}

   214→/* آخر جلسة محفوظة (لو الطالب فتح أكتر من غرفة بنآخذ الأحدث بالتوقيت) */
   215→function readSavedMe(): { code: string; playerId: string; token: string } | null {
   216→  try {
   217→    var st = battleStore()
   218→    if (!st) return null
   219→    var best: { code: string; playerId: string; token: string; ts: number } | null = null
   220→    for (var i = 0; i < st.length; i++) {
   221→      var k = st.key(i)
   222→      if (!k || k.indexOf(ME_KEY_PREFIX) !== 0) continue
   223→      var v = JSON.parse(st.getItem(k) || 'null')
   224→      if (v && v.playerId && v.token) {
   225→        var ts = Number(v.ts || 0)
   226→        if (!best || ts > best.ts) {
   227→          best = { code: k.slice(ME_KEY_PREFIX.length), playerId: String(v.playerId), token: String(v.token), ts: ts }
   228→        }
   229→      }
   230→    }
   231→    return best
   232→  } catch (e) { return null }
   233→}

   243→/* عداد تنازلي m:ss أو h:mm:ss — بيتعرض dir=ltr */
   244→function fmtCountdown(ms: number): string {
   245→  var s = Math.max(0, Math.floor(ms / 1000))
   246→  var h = Math.floor(s / 3600)
   247→  var m = Math.floor((s % 3600) / 60)
   248→  var sec = s % 60
   249→  var mm = (m < 10 ? '0' : '') + m
   250→  var ss = (sec < 10 ? '0' : '') + sec
   251→  return h > 0 ? (h + ':' + mm + ':' + ss) : (mm + ':' + ss)
   252→}

   709→  async function answerQuestion(i: number): Promise<void> {
   710→    var me = gMeRef.current
   711→    var room = gRoomRef.current
   712→    var qIdx = myQIdxRef.current
   713→    var qs = roomQRef.current
   714→    var q = qs[qIdx]
   715→    if (!me || !room || room.status !== 'live' || raceDoneRef.current || !q) return
   716→    if (sendingRef.current || myFeedback || shownIdxRef.current !== qIdx) return
   717→    var ms = myQElapsedMs()
   718→    setSendingAnswer(true)
   719→    setPendingChoice(i)
   720→    try {
   721→      var out = await postJson('/api/arena/rooms/' + encodeURIComponent(room.code), {
   722→        action: 'answer', playerId: me.id, token: me.token, questionIndex: qIdx, choice: i, ms: ms,
   723→      })
   724→      var data = out.data
   725→      if (out.status === 404) { resetGroups(); toast.error('الغرفة اتقفلت'); return }
   726→      if (out.status === 403) {
   727→        var msg403 = String((data && data.error) || 'جلسة غير صالحة')
   728→        resetGroups()
   729→        toast.error(msg403)
   730→        return
   731→      }
   732→      if (out.status === 409) {
   733→        /* الوقت خلص على السؤال ده — فيدباك ⌛ والتقدّم هيجي من السيرفر */
   734→        var msg = String((data && data.error) || 'الوقت خلص')
   735→        toast.error(msg + ' ⌛')
   736→        feedbackQIdxRef.current = qIdx
   737→        setMyFeedback({ correct: false, gained: 0, streak: 0, choice: i, timeout: true, questionIndex: qIdx })
   738→        scheduleAdvance(1100)
   739→        return
   740→      }
   741→      if (!data || !data.ok) { toast.error(String((data && data.error) || 'حصلت مشكلة في إجابتك')); return }
   742→      /* (و72) فيدباك لحظي ~1 ثانية → السؤال اللي بعده فورًا (من غير انتظار حد) */
   743→      feedbackQIdxRef.current = qIdx
   744→      setMyFeedback({ correct: !!data.correct, gained: Number(data.gained || 0), streak: Number(data.streak || 0), choice: i, correctIndex: Number(data.correctIndex), questionIndex: qIdx })
   745→      var nextIdx = Number(data.nextIndex != null ? data.nextIndex : qIdx + 1)
   746→      myQIdxRef.current = nextIdx
   747→      setMyQIdx(nextIdx)
   748→      var nq = qs[nextIdx]
   749→      var nl = nq ? Math.max(5, Number(nq.timeLimitSec || 25)) * 1000 : 30000
   750→      qRemainBaseRef.current = { remainMs: nl, at: Date.now(), limitMs: nl }
   751→      if (data.finished) scheduleFinish(950)
   752→      else scheduleAdvance(950)
   753→    } catch (e) {
   754→      toast.error('الشبكة بتلحس — جرب تاني')
   755→    } finally {
   756→      setSendingAnswer(false)
   757→      setPendingChoice(-1)
   758→    }
   759→  }
